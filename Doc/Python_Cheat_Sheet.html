📦 Module 01: Data Types & Math
ชนิดข้อมูลพื้นฐาน (Types) 01-Types
Typeคำอธิบายตัวอย่าง
intจำนวนเต็ม42, -10
floatจำนวนจริงทศนิยม3.14, 2.5e3
strข้อความสตริง"Hello"
boolค่าความจริงTrue, False
ตัวดำเนินการคณิตศาสตร์ 01-Operators
Opความหมายตัวอย่าง
+, -บวก, ลบ5 + 2 = 7
*, /คูณ, หาร (ได้ float)5 / 2 = 2.5
//หารปัดเศษลง5 // 2 = 2
%มอดุโล (เอาเศษ)5 % 2 = 1
**ยกกำลัง2 ** 3 = 8
โมดูล math 01-Math
import math
math.sqrt(25)       # 5.0
math.ceil(3.2)      # 4 (ปัดขึ้น)
math.floor(3.8)     # 3 (ปัดลง)
math.sin(math.radians(90)) # 1.0
math.pi             # 3.14159...
math.factorial(5)   # 120 (5!)
🔤 Module 02: Strings, Lists & Slicing
การตัดช่วง (Slicing) 02-Slice
s = "Python101"
s[0]       # 'P' (ตัวแรก)
s[-1]      # '1' (ตัวท้าย)
s[0:6]     # 'Python' (index 0..5)
s[:3]      # 'Pyt' (3 ตัวแรก)
s[3:]      # 'hon101' (ตั้งแต่ 3 จบ)
s[::-1]    # '101nohtyP' (กลับทิศ)
s[::2]     # 'Pto11' (ทีละ 2 สเต็ป)
Split & Join 02-Split
# แยกข้อความ (split)
words = "A,B,C,D".split(",")
# words: ['A', 'B', 'C', 'D']

# รวมข้อความ (join)
result = " - ".join(words)
# result: 'A - B - C - D'
⚖️ Module 03: Selection & Conditions
if - elif - else & Ternary 03-If
if score >= 80:
    grade = "A"
elif score >= 70:
    grade = "B"
else:
    grade = "F"

# If บรรทัดเดียว (Ternary)
status = "Pass" if score >= 50 else "Fail"
การตรวจสมาชิก (in / not in) 03-Membership
vowels = "aeiouAEIOU"
if "e" in vowels:
    print("เป็นสระ")

if "cat" not in ["dog", "bird"]:
    print("ไม่อยู่ในลิสต์")
🔁 Module 04: Loops & Repetition
While & For Loop 04-Loops
# while loop
n = 16
while n > 1:
    n //= 2

# for loop กับ range
for i in range(1, 6):     # 1, 2, 3, 4, 5
    print(i)

# break (หยุดลูป) / continue (ข้ามรอบ)
for x in range(10):
    if x == 5: break

⚙️ Module 06: Functions
Function & Return 06-Def
def calculate_stats(numbers):
    """ส่งคืนหลายค่าพร้อมกัน (Tuple)"""
    return min(numbers), max(numbers), sum(numbers)/len(numbers)

mn, mx, avg = calculate_stats([10, 20, 30])
📝 Module 07: String Processing & Files
String Methods & File I/O 07-Files
# String Methods
s = "  hello world  "
s.strip()            # "hello world"
s.replace("world", "Python")
s.find("lo")         # 3

# File I/O
with open("data.txt", "w", encoding="utf-8") as f:
    f.write("Line 1\n")

with open("data.txt", "r", encoding="utf-8") as f:
    for line in f:
        print(line.strip())
📖 Module 08: Dictionary
Key-Value & Frequency Counter 08-Dict
scores = {"Alice": 90, "Bob": 80}
scores["Charlie"] = 85

# Safe Get
print(scores.get("David", 0))  # 0

# Frequency Counter
freq = {}
for word in ["apple", "banana", "apple"]:
    freq[word] = freq.get(word, 0) + 1
# {'apple': 2, 'banana': 1}
🧩 Module 09: Nested & Comprehension
List Comprehension & 2D Matrix 09-Comprehension
# List Comprehension
evens_sq = [x**2 for x in range(10) if x % 2 == 0]

# สร้าง Matrix 3x4 ค่าเริ่มต้น 0
matrix = [[0 for _ in range(4)] for _ in range(3)]

# Matrix Transpose
A = [[1, 2], [3, 4], [5, 6]]
T = [[A[r][c] for r in range(len(A))] for c in range(len(A[0]))]
